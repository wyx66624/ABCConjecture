\section{Radical-excess obstructions to packet compression}
\label{sec:packet-radical-excess-obstruction}

\noindent\textbf{Author: ChatGPT.}
We continue with the synchronized divisor packets of
Section~\ref{sec:synchronized-divisor-packets}.  The purpose of this section
is twofold.  First, we identify the exact order-statistic geometry hidden in
the pair-max envelope.  Second, we correct the proposed radical-scale packet
gate by giving an infinite complete-premise counterfamily.  The
counterfamily disproves the packet gate, not the $abc$ conjecture.

Let
\[
 \mathcal R=\operatorname{rad}(abc),\qquad E=\frac{abc}{\mathcal R}.
\]
The integer $E$ will be called the radical excess.  Recall that every
synchronized packet $Q=(x,y,z)$ satisfies $abc\le B(Q)$.

\subsection{The shape identity}

Put
\[
 m(Q)=\min(x,y,z),\qquad T(Q)=\max(x,y,z),
\]
and
\[
 L(Q)=\max(y,z)\max(x,z)\max(x,y).
\]
Thus $B(Q)=L(Q)^2$.

\begin{theorem}[Symmetric packet-shape identity]
\label{thm:packet-shape-identity}
For every synchronized packet,
\[
 m(Q)L(Q)=xyzT(Q),
 \qquad
 m(Q)^2B(Q)=\bigl(xyzT(Q)\bigr)^2.
\]
\end{theorem}

\begin{proof}
Sort the coordinates as $r\le s\le t$.  Their pair maxima are $s,t,t$,
so $m=r$, $T=t$, and $L=st^2$.  Consequently
$mL=rst^2=(rst)t=xyzT$.  Squaring proves the second identity.  Since both
sides are symmetric, the computation is independent of the sorting
permutation.
\end{proof}

\begin{proposition}[Support and full-envelope monotonicity]
\label{prop:packet-support-envelope}
Every synchronized packet satisfies
\[
 xyz\mid abc,\qquad
 \operatorname{rad}(xyz)\mid \mathcal R,\qquad
 B(Q)\le B(a,b,c).
\]
\end{proposition}

\begin{proof}
Multiply the three coordinate divisibilities.  Radical divisibility is
monotone under divisibility for nonzero natural numbers.  Finally,
$x\le a$, $y\le b$, and $z\le c$; hence every pair maximum for $Q$ is no
larger than its full-packet counterpart.  Square and multiply.
\end{proof}

Equality of coordinate support with full radical support is false.  The
complete-premise packet $(5,7,2)$ over $(5,7,12)$ has coordinate radical
$70$, whereas $\operatorname{rad}(5\cdot7\cdot12)=210$.

\subsection{The radical-excess obstruction}

\begin{theorem}[Necessary condition for rational packet compression]
\label{thm:packet-radical-excess-necessary}
Let $m,n\ge0$.  If
\[
 B(Q)^m\le \mathcal R^{m+n},
\]
then
\[
 E^m\le \mathcal R^n.
\]
Equivalently, $\mathcal R^n<E^m$ implies
$\mathcal R^{m+n}<B(Q)^m$ for every synchronized
packet $Q$ over the same primitive triple.
\end{theorem}

\begin{proof}
Since $\mathcal R\mid abc$, one has $\mathcal R E=abc$.  Therefore
\[
 \mathcal R^mE^m=(abc)^m\le B(Q)^m
 \le \mathcal R^{m+n}=\mathcal R^m\mathcal R^n.
\]
The radical is positive, so cancellation of $\mathcal R^m$ gives the assertion.
\end{proof}

This condition depends only on the underlying triple.  In particular, no
choice inside the packet spectrum can repair its failure.

\begin{theorem}[Infinite four-thirds obstruction]
\label{thm:packet-dyadic-four-thirds}
For $k\ge0$, set
\[
 A_k=2^{k+4},\qquad P_k=(A_k,3,A_k+3).
\]
These are pairwise distinct primitive nonunit triples.  If
$\mathcal R_k=\operatorname{rad}(A_k\cdot3\cdot(A_k+3))$, then every synchronized
packet over $P_k$ satisfies
\[
 \mathcal R_k^4<B(Q)^3.
\]
Consequently the locus on which every packet violates
$B^3\le \mathcal R^4$ is
infinite.
\end{theorem}

\begin{proof}
Put $h=2^{k+3}$, so $A_k=2h$ and $h\ge8$.  The triple is primitive because
$A_k$ is a power of $2$ and is coprime to $3$.  Radical
submultiplicativity gives
\[
 \mathcal R_k\le
 \operatorname{rad}(A_k)\operatorname{rad}(3)
 \operatorname{rad}(A_k+3)
 \le 2\cdot3\cdot(A_k+3)=6(2h+3).
\]
Writing $E_k=A_k3(A_k+3)/\mathcal R_k$, the equality
$\mathcal R_kE_k=6h(A_k+3)$ and the preceding upper bound imply $h\le E_k$.
Moreover,
\[
 \mathcal R_k\le 6(2h+3)=12h+18<64h\le h^3\le E_k^3.
\]
Theorem~\ref{thm:packet-radical-excess-necessary}, with $(m,n)=(3,1)$,
now gives $\mathcal R_k^4<B(Q)^3$ for every packet.  The map $k\mapsto P_k$ is
injective because its first coordinate is strictly increasing.
\end{proof}

For positive real $\mathcal R$ and $B$, an estimate
$B\le \mathcal R^{4/3}$ implies
$B^3\le \mathcal R^4$.  Theorem~\ref{thm:packet-dyadic-four-thirds} therefore
refutes the earlier all-but-finitely-many packet gate
$B(Q)\le \mathcal R^{1+\varepsilon}$ at the exact value
$\varepsilon=1/3$.  It does not produce triples violating
$c\le \mathcal R^{1+\varepsilon}$, and hence gives no
counterexample to $abc$.

\subsection{A correctly compensated forward gate}

\begin{theorem}[Compensated packet implication]
\label{thm:packet-compensated-implication}
Let $m,n\ge0$.  If
\[
 B(Q)^m\le (ab)^m\mathcal R^{m+n},
\]
then
\[
 c^m\le \mathcal R^{m+n}.
\]
\end{theorem}

\begin{proof}
The synchronized product bound gives
\[
 (ab)^mc^m=(abc)^m\le B(Q)^m
 \le (ab)^m\mathcal R^{m+n}.
\]
Cancel the positive factor $(ab)^m$.
\end{proof}

Thus the factor $ab$ is the exact elementary compensator that changes a
packet bound for the full product into the desired bound for the sum arm.
No eventual compensated estimate is proved here.  Its pointwise
four-thirds form already fails at $(2,3,5)$, whose packet spectrum consists
only of the full packet.  The all-but-finitely-many compensated direction
remains open because this section supplies no infinite counterfamily to it.
